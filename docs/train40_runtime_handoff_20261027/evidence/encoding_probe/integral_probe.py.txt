"""Unadmitted rolling integer sensor moments, including one actual 20ms update.

All histogram work is causal. Initial history contains only samples before
query-20ms; the timed update ingests the final raw 20ms CPU packet, changes
the query ROI, and returns all six voxel windows on CUDA. Disk replay before
CPU packet availability is declared separately and excluded from that clock.
"""
from __future__ import annotations

import argparse
import json
import sys
import time
from pathlib import Path

ROOT=Path(__file__).resolve().parents[3]
sys.path[:0]=[str(ROOT/'src'),str(ROOT)]
import numpy as np
import torch
from e_jepa_ttc.data.eap import EAP_IMAGE_SIZE
from e_jepa_ttc.efficient_context.mapped_union import encode_mapped,map_roi
from operational.evttc_transfer.inputs import EvTTCEventReader


def normalize_grid(voxel,count,size):
    voxel=voxel.reshape(6,-1)
    occupied=voxel.abs()>1e-6
    nonzero=occupied.sum(1)
    ordered=voxel.abs().sort(dim=1).values
    rank=(nonzero-1).clamp_min(0).double()*.95
    lower=rank.floor().long()
    upper=rank.ceil().long()
    offset=voxel.shape[1]-nonzero
    a=ordered.gather(1,(offset+lower).clamp_max(voxel.shape[1]-1)[:,None])[:,0]
    b=ordered.gather(1,(offset+upper).clamp_max(voxel.shape[1]-1)[:,None])[:,0]
    fraction=rank-lower
    scale=torch.where(fraction>=.5,b-(b-a)*(1.-fraction),a+(b-a)*fraction).clamp_min(1e-6)
    grid=torch.where(occupied,voxel/scale[:,None],0.).float().reshape(6,10,size,size)
    scalars=torch.stack((torch.log1p(count.double()),torch.log1p(count.double()/(100000*1e-6))),1).float()
    return torch.cat((grid,scalars[:,:,None,None].expand(-1,-1,size,size)),1)


class IntegralMomentHistory:
    def __init__(self,start,size=128):
        self.width,self.height=EAP_IMAGE_SIZE
        self.size=size
        self.end=start
        self.oldest=0
        self.filled=0
        shape=(120,2,self.height+1,self.width+1)
        self.counts=torch.zeros(shape,dtype=torch.int32,device='cuda')
        self.moments=torch.zeros(shape,dtype=torch.int64,device='cuda')
        self.offsets=torch.arange(5,device='cuda',dtype=torch.int64).reshape(1,5,1,1,1)*5000

    def ingest(self,raw,start,end):
        if start!=self.end or (end-start)%5000:
            raise ValueError('noncontiguous or non5ms stream')
        values={k:torch.from_numpy(v).to('cuda') for k,v in raw.items()}
        xy=values['y'].long()*self.width+values['x'].long()
        flat=xy+(values['p']<=0).long()*self.width*self.height
        for left in range(start,end,5000):
            lo,hi=np.searchsorted(raw['t'],[left,left+5000],side='left')
            if hi-lo>=2**31:
                raise ValueError('int32 count prefix capacity')
            slot=self.oldest
            self.oldest=(self.oldest+1)%120
            indices=flat[lo:hi]
            counts=torch.zeros(2*self.height*self.width,dtype=torch.int32,device='cuda')
            moments=torch.zeros(2*self.height*self.width,dtype=torch.int64,device='cuda')
            counts.scatter_add_(0,indices,torch.ones_like(indices,dtype=torch.int32))
            moments.scatter_add_(0,indices,values['t'][lo:hi]-left)
            counts=counts.reshape(2,self.height,self.width).cumsum(1,dtype=torch.int32).cumsum(2,dtype=torch.int32)
            moments=moments.reshape(2,self.height,self.width).cumsum(1).cumsum(2)
            self.counts[slot,:,1:,1:].copy_(counts)
            self.moments[slot,:,1:,1:].copy_(moments)
            self.filled=min(120,self.filled+1)
        self.end=end

    @staticmethod
    def bounds(length,low,high,size):
        axis=np.arange(length,dtype=np.float64)
        valid=(axis>=low)&(axis<high)
        positions=np.flatnonzero(valid)
        if not len(positions):
            return np.zeros(size+1,dtype=np.int64)
        projected=np.clip(np.floor((axis[valid]-low)*size/(high-low)).astype(np.int32),0,size-1)
        return np.searchsorted(projected,np.arange(size+1),side='left')+positions[0]

    def query(self,roi):
        if self.filled!=120:raise ValueError('history is not filled')
        x0,y0,x1,y1=roi
        xb=torch.from_numpy(self.bounds(self.width,x0,x1,self.size)).to('cuda')
        yb=torch.from_numpy(self.bounds(self.height,y0,y1,self.size)).to('cuda')
        # All four corner arrays contain only128*128 positions. The temporal
        # ring remains independent of the current query's ROI.
        corners=[(yb[yi:yi+self.size,None]*(self.width+1)+xb[None,xi:xi+self.size]).reshape(-1)
                 for yi,xi in ((1,1),(0,1),(1,0),(0,0))]
        order=(torch.arange(120,device='cuda')+self.oldest)%120
        values=[]
        for source in (self.counts,self.moments):
            flat=source.reshape(120,2,-1)
            selected=flat.index_select(2,corners[0])-flat.index_select(2,corners[1])
            selected-=flat.index_select(2,corners[2])
            selected+=flat.index_select(2,corners[3])
            values.append(selected.index_select(0,order).reshape(24,5,2,self.size,self.size))
        c,s=values
        quarters=c.sum(1)
        numerator=(s+c.to(torch.int64)*self.offsets).sum(1)
        upper=numerator.double()/25000.
        lower=quarters.double()-upper
        upper=upper.reshape(6,4,2,self.size,self.size).transpose(1,2)
        lower=lower.reshape(6,4,2,self.size,self.size).transpose(1,2)
        voxel=torch.zeros((6,2,5,self.size,self.size),device='cuda',dtype=torch.float64)
        voxel[:,:,:4]=lower
        voxel[:,:,1:]+=upper
        count=quarters.reshape(6,4,2,self.size,self.size).sum((1,2,3,4))
        return normalize_grid(voxel,count,self.size)


def main():
    parser=argparse.ArgumentParser()
    parser.add_argument('--indices',nargs='+',type=int,default=[0,200,553])
    parser.add_argument('--repeats',type=int,default=8)
    args=parser.parse_args()
    torch.set_num_threads(2)
    rows=json.loads((ROOT/'artifacts/final_train40_transfer_20261027/QUERY_MANIFEST.json').read_text(encoding='utf-8'))['rows']
    records=[]
    for index in args.indices:
        row=rows[index]
        anchor=int(row['anchor_us'])
        square=tuple(row['square_xyxy'])
        with EvTTCEventReader(row['raw_path']) as reader:
            startup=reader.read_window(anchor-620000,anchor-20000)
            packet=reader.read_window(anchor-20000,anchor)
        raw={k:np.concatenate((startup[k],packet[k])) for k in startup}
        mapped=map_roi(raw,square,128,0.)
        reference=np.stack([encode_mapped(mapped,left,left+100000,128,'probe').numpy() for left in range(anchor-600000,anchor,100000)])
        del raw,mapped
        begin=time.perf_counter()
        runtime=IntegralMomentHistory(anchor-620000)
        runtime.ingest(startup,anchor-620000,anchor-20000)
        torch.cuda.synchronize()
        startup_ms=(time.perf_counter()-begin)*1000
        # Warm only query-shaped kernels on the available historical state.
        for _ in range(3):runtime.query(square)
        torch.cuda.synchronize()
        begin=time.perf_counter()
        runtime.ingest(packet,anchor-20000,anchor)
        output=runtime.query(square)
        torch.cuda.synchronize()
        update_ms=(time.perf_counter()-begin)*1000
        actual=output.cpu().numpy()
        query_ms=[]
        for _ in range(args.repeats):
            torch.cuda.synchronize()
            begin=time.perf_counter()
            runtime.query(square)
            torch.cuda.synchronize()
            query_ms.append((time.perf_counter()-begin)*1000)
        difference=np.abs(actual-reference)
        record={'index':index,'startup_ms':startup_ms,'startup_events':len(startup['t']),
                'packet_events':len(packet['t']),'actual20ms_ingest_and_current_roi_ms':update_ms,
                'query_only_ms':query_ms,'equal':bool(np.array_equal(actual,reference)),
                'different_values':int(np.count_nonzero(difference)),'max_abs':float(difference.max()),
                'allclose_1e5':bool(np.allclose(actual,reference,rtol=1e-5,atol=1e-5)),
                'peak_cuda_bytes':torch.cuda.max_memory_allocated()}
        records.append(record)
        print(json.dumps(record),flush=True)
        del runtime,startup,packet,reference,actual,output
        torch.cuda.empty_cache()
    (Path(__file__).parent/'integral_results.json').write_text(json.dumps({'status':'UNADMITTED_INTEGER_MOMENT_REORDERING','scope':'CPU raw packet availability to GPU voxel, model excluded','records':records},indent=2))


if __name__=='__main__':main()
