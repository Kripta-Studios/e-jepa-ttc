"""UNADMITTED GPU resident raw voxel experiment; excludes model and initial IO."""
from __future__ import annotations

import argparse
import json
import sys
import time
from pathlib import Path

ROOT = Path(__file__).resolve().parents[3]
sys.path[:0] = [str(ROOT / 'src'), str(ROOT)]
import numpy as np
import torch

from e_jepa_ttc.data.eap import EAP_IMAGE_SIZE
from e_jepa_ttc.efficient_context.mapped_union import encode_mapped, map_roi
from operational.evttc_transfer.inputs import EvTTCEventReader


class ResidentVoxelPrototype:
    def __init__(self, raw, start, end, size=128, *, moments=False):
        self.start, self.end, self.size = start,end,size
        self.moments=moments
        self.raw = {k:torch.from_numpy(v).to('cuda') for k,v in raw.items()}
        self.elapsed = self.raw['t']-start
        self.window = torch.div(self.elapsed,100000,rounding_mode='floor')
        self.relative = self.elapsed % 100000
        self.polarity = (self.raw['p']<=0).to(torch.int64)
        self.payload_size = 6*10*size*size
        self.scratch_indices = torch.arange(len(raw['t']),device='cuda')+self.payload_size
        self.xlut=torch.empty(EAP_IMAGE_SIZE[0],dtype=torch.int32,device='cuda')
        self.ylut=torch.empty(EAP_IMAGE_SIZE[1],dtype=torch.int32,device='cuda')
        times=np.arange(100000,dtype=np.float64)/100000.*4
        lower=np.floor(times).astype(np.int64)
        upper=np.ceil(times).astype(np.int64)
        uw=times-lower
        self.lower=torch.from_numpy(lower).to('cuda')
        self.upper=torch.from_numpy(upper).to('cuda')
        self.lw=torch.from_numpy(1.-uw).to('cuda')
        self.uw=torch.from_numpy(uw).to('cuda')

    def set_roi(self,roi):
        x0,y0,x1,y1=roi
        xt=np.arange(EAP_IMAGE_SIZE[0],dtype=np.float64)
        yt=np.arange(EAP_IMAGE_SIZE[1],dtype=np.float64)
        xlut=np.clip(np.floor((xt-x0)*self.size/(x1-x0)).astype(np.int32),0,self.size-1)
        ylut=np.clip(np.floor((yt-y0)*self.size/(y1-y0)).astype(np.int32),0,self.size-1)
        xlut[(xt<x0)|(xt>=x1)]=-1
        ylut[(yt<y0)|(yt>=y1)]=-1
        self.xlut.copy_(torch.from_numpy(xlut))
        self.ylut.copy_(torch.from_numpy(ylut))

    def compute(self):
        size=self.size
        x=self.xlut[self.raw['x'].long()]
        y=self.ylut[self.raw['y'].long()]
        valid=(x>=0)&(y>=0)
        count=torch.stack([(valid & (self.window==i)).sum() for i in range(6)])
        spatial=y.to(torch.int64)*size+x
        base=(self.window*10+self.polarity*5)*size*size+spatial
        lower_idx=base+self.lower[self.relative]*size*size
        upper_idx=base+self.upper[self.relative]*size*size
        lower_idx=torch.where(valid,lower_idx,self.scratch_indices)
        upper_idx=torch.where(valid,upper_idx,self.scratch_indices)
        output_size=self.payload_size+len(self.raw['t'])
        lower=torch.zeros(output_size,dtype=torch.float64,device='cuda')
        upper=torch.zeros(output_size,dtype=torch.float64,device='cuda')
        lower.scatter_add_(0,lower_idx,self.lw[self.relative])
        upper.scatter_add_(0,upper_idx,self.uw[self.relative])
        voxel=(lower[:self.payload_size]+upper[:self.payload_size]).reshape(6,-1)
        occupied=voxel.abs()>1e-6
        nonzero=occupied.sum(1)
        ordered=voxel.abs().sort(dim=1).values
        rank=(nonzero-1).clamp_min(0).double()*.95
        rank_lower=rank.floor().long()
        rank_upper=rank.ceil().long()
        offset=voxel.shape[1]-nonzero
        at_lower=(offset+rank_lower).clamp_max(voxel.shape[1]-1)
        at_upper=(offset+rank_upper).clamp_max(voxel.shape[1]-1)
        a=ordered.gather(1,at_lower[:,None])[:,0]
        b=ordered.gather(1,at_upper[:,None])[:,0]
        fraction=rank-rank_lower
        scale=torch.where(fraction>=.5,b-(b-a)*(1.-fraction),a+(b-a)*fraction).clamp_min(1e-6)
        grid=torch.where(occupied,voxel/scale[:,None],0.).float().reshape(6,10,size,size)
        scalars=torch.stack((torch.log1p(count.double()),torch.log1p(count.double()/.1)),1).float()
        scalar_grid=scalars[:,:,None,None].expand(-1,-1,size,size)
        return torch.cat((grid,scalar_grid),1)


def timing(fn,repeats):
    values=[]
    out=None
    for _ in range(repeats):
        torch.cuda.synchronize()
        begin=time.perf_counter()
        out=fn()
        torch.cuda.synchronize()
        values.append(1000*(time.perf_counter()-begin))
    return out,values


def main():
    parser=argparse.ArgumentParser()
    parser.add_argument('--indices',nargs='+',type=int,default=[0,200,553])
    parser.add_argument('--repeats',type=int,default=5)
    args=parser.parse_args()
    torch.set_num_threads(2)
    rows=json.loads((ROOT/'artifacts/final_train40_transfer_20261027/QUERY_MANIFEST.json').read_text(encoding='utf-8'))['rows']
    records=[]
    for index in args.indices:
        row=rows[index]
        start=int(row['anchor_us'])-600000
        end=int(row['anchor_us'])
        square=tuple(row['square_xyxy'])
        with EvTTCEventReader(row['raw_path']) as reader:
            raw=reader.read_window(start,end)
        assert (raw['x']>=0).all() and (raw['x']<1280).all()
        assert (raw['y']>=0).all() and (raw['y']<720).all()
        mapped=map_roi(raw,square,128,0.)
        reference=np.stack([encode_mapped(mapped,left,left+100000,128,'probe').numpy() for left in range(start,end,100000)])
        runtime,init_ms=timing(lambda:ResidentVoxelPrototype(raw,start,end),1)
        runtime.set_roi(square)
        actual,warm_ms=timing(runtime.compute,1)
        actual,wall_ms=timing(runtime.compute,args.repeats)
        actual=actual.cpu().numpy()
        difference=np.abs(actual-reference)
        # Capture only fixed raw count/timestamps as a prepared-history probe.
        stream=torch.cuda.Stream()
        stream.wait_stream(torch.cuda.current_stream())
        with torch.cuda.stream(stream):
            for _ in range(3):runtime.compute()
        torch.cuda.current_stream().wait_stream(stream)
        torch.cuda.synchronize()
        graph=torch.cuda.CUDAGraph()
        with torch.cuda.graph(graph):graph_out=runtime.compute()
        _,graph_ms=timing(graph.replay,args.repeats)
        graph_array=graph_out.cpu().numpy()
        graph_diff=np.abs(graph_array-reference)
        record={'index':index,'raw_events':len(raw['t']),'mapped_events':len(mapped['t']),
                'initial_h2d_and_setup_ms':init_ms,'cold_compute_ms':warm_ms,'eager_ms':wall_ms,
                'graph_ms':graph_ms,'equal':bool(np.array_equal(actual,reference)),
                'different_values':int(np.count_nonzero(difference)),
                'max_abs':float(difference.max()),'allclose_1e5':bool(np.allclose(actual,reference,rtol=1e-5,atol=1e-5)),
                'graph_equal':bool(np.array_equal(graph_array,reference)),
                'graph_max_abs':float(graph_diff.max()),'peak_allocated':torch.cuda.max_memory_allocated()}
        records.append(record)
        print(json.dumps(record),flush=True)
        del graph,graph_out,runtime,raw,mapped,actual,reference
        torch.cuda.empty_cache()
    (Path(__file__).parent/'gpu_results.json').write_text(json.dumps({'status':'UNADMITTED_RESIDENT_RAW_PROBE_EXCLUDES_NEW_DATA','records':records},indent=2))


if __name__=='__main__':main()
