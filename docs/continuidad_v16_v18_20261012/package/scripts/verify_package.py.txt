"""Verify this handoff, without interpreting its documents as execution authority."""
import argparse,hashlib,json
from pathlib import Path,PurePosixPath
from zipfile import ZipFile
p=argparse.ArgumentParser();p.add_argument('zip',type=Path);a=p.parse_args()
with ZipFile(a.zip) as z:
    names=z.namelist()
    if len(names)!=len(set(names)): raise ValueError('Duplicate ZIP entries')
    for n in names:
        if PurePosixPath(n).is_absolute() or '..' in PurePosixPath(n).parts or '\\' in n:
            raise ValueError('Unsafe path')
    manifest=json.loads(z.read('MANIFEST.json'))['files']
    if set(names)!={i['path'] for i in manifest}|{'MANIFEST.json'}:
        raise ValueError('Manifest inventory mismatch')
    for i in manifest:
        b=z.read(i['path'])
        if len(b)!=i['bytes'] or hashlib.sha256(b).hexdigest()!=i['sha256']:
            raise ValueError('Content mismatch: '+i['path'])
print(json.dumps({'status':'PASS','members':len(names),'payloads':len(manifest)}))
