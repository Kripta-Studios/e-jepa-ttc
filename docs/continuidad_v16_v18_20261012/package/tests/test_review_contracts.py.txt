import sys,unittest,math
from pathlib import Path
sys.path.insert(0,str(Path(__file__).resolve().parents[1]/'scripts'))
from review_contracts import phase,row_mid,partial_positive,validate_results
from check_submission import no_duplicate_keys
class Checks(unittest.TestCase):
    def test_signed_domains(self):
        self.assertGreater(phase(2),0);self.assertLess(phase(-2),0)
    def test_bad_domain(self):
        for x in [0,.1,.05,math.nan,math.inf]:
            with self.subTest(x=x),self.assertRaises(ValueError):phase(x)
    def test_identity(self):self.assertEqual(row_mid(2,2),0)
    def test_symmetry(self):self.assertEqual(row_mid(2,3),row_mid(3,2))
    def test_negative_is_not_invalid(self):self.assertTrue(math.isfinite(row_mid(-2,2)))
    def test_partial_does_not_invent_overall(self):
        r=partial_positive({'c':100,'s':30,'l':10});self.assertEqual(r['observed_positive_contribution'],60)
        self.assertIsNone(r['official_overall_MiD']);self.assertIsNone(r['negative_contribution'])
    def test_missing_band(self):
        with self.assertRaises(ValueError):partial_positive({'c':100,'s':30})
    def test_submission_counts(self):
        p={'meta':{'format':'garlttc_prediction_v1'},'results':{'x':{'ttc':-2}}}
        self.assertEqual(validate_results(p,1),{'tokens':1})
        with self.assertRaises(ValueError):validate_results(p,2)
    def test_duplicate_json(self):
        with self.assertRaises(ValueError):no_duplicate_keys([('x',1),('x',2)])
if __name__=='__main__':unittest.main()
