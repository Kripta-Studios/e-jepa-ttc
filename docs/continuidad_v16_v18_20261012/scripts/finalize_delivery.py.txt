def pack():
    downloads=Path.home()/'Downloads'
    blind=OUT/'PAQUETE_ANOTACION_CIEGA.zip'
    with zipfile.ZipFile(blind,'w',compression=zipfile.ZIP_DEFLATED) as archive:
        for name in ('ANNOTATION_TEMPLATE.csv','INSTRUCCIONES_CIEGAS.md'):
            archive.write(OUT/'geometry'/name,name)
        for path in sorted((OUT/'geometry/panels').glob('*.png')):
            archive.write(path,'panels/'+path.name)
    files=[]
    for path in sorted(OUT.rglob('*')):
        if path.is_file() and '__pycache__' not in path.parts and path.name!='DELIVERY_MANIFEST.json' and path.suffix!='.pending':
            files.append({'path':path.relative_to(OUT).as_posix(),'bytes':path.stat().st_size,'sha256':sha(path)})
    write(OUT/'DELIVERY_MANIFEST.json',{'schema':'sha256_delivery_v1','utc':datetime.now(timezone.utc).isoformat(),'files':files})
    archive_path=downloads/'E_JEPA_TTC_CONTINUIDAD_EJECUTADA_20261012.zip'
    if archive_path.exists(): raise FileExistsError('Delivery already exists; do not overwrite: '+str(archive_path))
    with zipfile.ZipFile(archive_path,'w',compression=zipfile.ZIP_DEFLATED) as archive:
        for row in files: archive.write(OUT/row['path'],row['path'])
        archive.write(OUT/'DELIVERY_MANIFEST.json','DELIVERY_MANIFEST.json')
    with tempfile.TemporaryDirectory(prefix='continuidad_verify_') as temp:
        directory=Path(temp)
        with zipfile.ZipFile(archive_path) as archive:
            assert archive.testzip() is None
            assert set(archive.namelist())=={r['path'] for r in files}|{'DELIVERY_MANIFEST.json'}
            archive.extractall(directory)
        expected=read(directory/'DELIVERY_MANIFEST.json')
        for row in expected['files']:
            path=directory/row['path']; assert path.stat().st_size==row['bytes'] and sha(path)==row['sha256']
        extracted_tests=[]
        python=WORKSPACE/'e-jepa-ttc/.venv/Scripts/python.exe'
        commands=[('package_contracts',9,directory/'package',['-m','unittest','discover','-s','tests','-v']),('RGB_cache',6,directory,['scripts/test_rgb_exact_cache.py','-v']),('V18_diagnostics_synthetic',2,directory,['scripts/test_v18_diagnostics.py','-v'])]
        for name,count,cwd,arguments in commands:
            result=subprocess.run([str(python),'-B',*arguments],cwd=cwd,capture_output=True,text=True,encoding='utf-8',errors='replace',check=False)
            log=result.stdout+'\n'+result.stderr
            assert result.returncode==0 and f'Ran {count} tests' in log and '\nOK' in log,log
            extracted_tests.append({'suite':name,'tests':count,'exit_code':result.returncode,'output':log})
    receipt={'status':'PASS','archive':str(archive_path),'bytes':archive_path.stat().st_size,'sha256':sha(archive_path),'payload_files':len(files),'crc':'PASS','independent_extraction_sha256':'PASS','independent_extraction_tests':extracted_tests,'candidate_sha256':'60b5a620848659633b875772430cddd7d7532b0fe180ec59904d04b81a9d3ca1','official_submission':False,'utc':datetime.now(timezone.utc).isoformat()}
    write(downloads/'E_JEPA_TTC_CONTINUIDAD_EJECUTADA_20261012.verification.json',receipt)
    print(json.dumps(receipt,indent=2))
