"""Check signed/capped diagnostics and refuse unsealed prediction changes."""
import hashlib
import json
import tempfile
import unittest
from pathlib import Path
from unittest.mock import patch

import numpy as np
import pandas as pd

import diagnose_v18_predictions as diagnostic


class SealedDiagnostics(unittest.TestCase):
    def fixture(self,folder,fit):
        root=Path(folder)/'run'; out=Path(folder)/'output'; out.mkdir(exist_ok=True)
        directory=root/'predictions/H'/fit; directory.mkdir(parents=True)
        frame=pd.DataFrame({'sample_token':['a','b','c'],'group_id':['g1','g1','g2'],'sequence_id':['s1','s1','s2'],'target_ttc':[1.5,-2.,4.],'prediction_ttc':[60.,60.,4.],'raw_phase':[0.,0.,-np.log1p(-.1/4)]})
        path=directory/'predictions.parquet'; frame.to_parquet(path,index=False)
        (directory/'COMPLETE.json').write_text(json.dumps({'status':'COMPLETE','sha256':hashlib.sha256(path.read_bytes()).hexdigest()}))
        return root,out,path

    def test_signed_caps_and_separate_group_sequence_units(self):
        with tempfile.TemporaryDirectory() as folder:
            root,out,_=self.fixture(folder,'PAIR-DIRECT-S7')
            with patch.object(diagnostic,'RUN',root),patch.object(diagnostic,'OUT',out): diagnostic.main()
            result=pd.read_csv(out/'v18_diagnostics/DIAGNOSTICO_SIGNO_RANGO.csv')
            negative=result[(result.unit=='all')&(result.band=='negative')].iloc[0]
            self.assertEqual(negative.wrong_sign,1)
            self.assertEqual(negative.bias,62.)
            self.assertEqual(negative.cap60,1)
            self.assertEqual(negative.flat_decoder_region_count,1)
            self.assertEqual(negative.raw_emitted_sign_changes,1)
            self.assertEqual(set(result.unit),{'all','group_id','sequence_id'})

    def test_native_raw_not_misrepresented_and_changed_seal_rejected(self):
        with tempfile.TemporaryDirectory() as folder:
            root,out,path=self.fixture(folder,'NATIVE-P2-FT-S7')
            with patch.object(diagnostic,'RUN',root),patch.object(diagnostic,'OUT',out):
                diagnostic.main()
                result=pd.read_csv(out/'v18_diagnostics/DIAGNOSTICO_SIGNO_RANGO.csv')
                self.assertTrue(result.raw_phase_mean.isna().all())
                self.assertTrue(result.flat_decoder_region_count.isna().all())
                with path.open('ab') as stream: stream.write(b'changed')
                with self.assertRaisesRegex(ValueError,'seal mismatch'): diagnostic.main()


if __name__=='__main__':unittest.main()
