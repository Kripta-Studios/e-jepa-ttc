"""Publish a passive handoff, candidate registry and reproduction comparisons."""
import csv
from datetime import datetime, timezone

import numpy as np
import pandas as pd

from prepare_delivery import OUT, PREVIOUS, RESEARCH, WORKSPACE, copy_same, read, sha, write


def main():
    records=[]
    v13=WORKSPACE/'e-jepa-ttc-v13-rgb-port/artifacts/rgb_port_20261008'
    train40=WORKSPACE/'e-jepa-ttc-v12-efficient-context/artifacts/train40_system_20261005'
    def add(name, role, modality, path, contract, fallback, official='NOT_SUBMITTED'):
        assert path.is_file(), path
        records.append(dict(model=name,role=role,modality=modality,identity_file=str(path),identity_sha256=sha(path),contract=contract,fallback=fallback,official_status=official))
    add('PHASE_ONLY_EVENT_S7','frozen_test12_candidate','RGB+Event',RESEARCH/'test12_phase_only/MODEL_INPUT_FREEZE.json','V13 frozen parents; 1955 new parameters; seed7 head; internal V selection; no TRAIN40 refit','E_CTX_MATCHED if RGB unavailable; test12 none')
    for name,modality,role in [('F_TRUE','RGB+Event','transfer_reference'),('R_CTX','RGB','incremental_event_reference'),('E_CTX_MATCHED','Event','matched_event_reference')]:
        add(name,role,modality,v13/f'fits/{name}/checkpoint_last.pt','V13 parents; distinct training/context from TRAIN40 H8','E_CTX_MATCHED for F_TRUE; none for standalone references')
    for seed in (7,13,23):
        add(f'H8_V12_S{seed}','event_only_reference','Event',train40/f'fits/h8_seed{seed}/checkpoint_last.pt','TRAIN40-system H8; 8 positions separated by 50ms, 350ms anchors; frozen earlier parents; not matched V13 pool','none')
    add('Garl_event','external_native_reference','Event',train40/'DELIVERY_FREEZE.json','Released event checkpoint; native official preprocessing; two endpoint windows','none','NO_CANDIDATE_SCORE')
    for name in ('full','visual'):
        add(f'Garl_{name}','external_native_reference','RGB+Event' if name=='full' else 'RGB',PREVIOUS/f'garl_{name}/BINDING.json','Released checkpoint, native input; estimated FCWD registration; common RGB coverage472 labeled','Garl event for full system; visual reported on available RGB','NO_CANDIDATE_SCORE')
    with (OUT/'REGISTRO_CANDIDATOS.csv').open('w',newline='',encoding='utf-8') as stream:
        writer=csv.DictWriter(stream,fieldnames=list(records[0])); writer.writeheader(); writer.writerows(records)
    checks=[]
    for name in ('BANDS.csv','GAP_DECOMPOSITION.csv','SIGN_ERROR_SHARE.csv'):
        a=pd.read_csv(OUT/'package/evidence'/name); b=pd.read_csv(OUT/'diagnostics'/name)
        assert list(a)==list(b) and a.shape==b.shape
        maximum=0.
        for col in a:
            if pd.api.types.is_numeric_dtype(a[col]):
                np.testing.assert_allclose(a[col],b[col],rtol=0,atol=1e-9,equal_nan=True)
                d=np.abs(a[col]-b[col]); maximum=max(maximum,float(d.max()) if d.notna().any() else 0.)
            else: assert a[col].fillna('').equals(b[col].fillna(''))
        checks.append({'table':name,'rows':len(a),'max_abs_numeric_difference':maximum,'status':'PASS'})
    write(OUT/'verification/CONTINUITY_TABLE_PARITY.json',{'tolerance_absolute':1e-9,'relative':0,'checks':checks})
    shared=WORKSPACE/'E_JEPA_SHARED_COORDINATION'
    copy_same(OUT/'package/MENSAJE_V18.md',shared/'V16_V18_CONTINUIDAD_20261012_MENSAJE.md')
    request={'schema':'v16_v18_descriptive_diagnostics_request_v1','utc':datetime.now(timezone.utc).isoformat(),'owner':'V16 continuation; passive handoff only','delivery':'shared_file_written; no live Codex session message API available; acknowledgement pending','V18_should_not_restart_or_change_active_recipe':True,'new_budget':0,'V16_optimizer_updates':0,'diagnostics_requested_after_start':True,'wait_for_primary_endpoint_freeze_before_H':True,'roles':['P','H','V'],'diagnostics':['signed bias and MAE: 0.1-1,1-2,2-3,3-6,6-10,negative','sign confusion, raw/emitted phase, saturation, flat decoder regions','group and sequence/episode results; P fit separated from H screening and exposed V development'],'native_phase_gap':'Native evaluate.py currently reconstructs raw_phase from final TTC; true predecoder phase must be observed by V18 owner in separately registered diagnostic inference, not fabricated here. No active fit change requested.','read_only_postprocessor':str(OUT/'scripts/diagnose_v18_predictions.py'),'evidence':str(OUT),'handoff':str(shared/'V16_V18_CONTINUIDAD_20261012_MENSAJE.md'),'official_submission':'manual user only; agent forbidden by latest user message','FCWD_Dev32_role':'descriptive motivation only; not recipe or checkpoint selection','model_changes_require_rebuild_of_downstream_dependencies':True}
    path=shared/'V16_V18_CONTINUIDAD_20261012_REQUEST.json'
    if not path.exists(): write(path,request)
    copy_same(path,OUT/'V18_COORDINATION_REQUEST.json')
    print({'registry_rows':len(records),'table_checks':checks,'V18_ack':'PENDING'})


if __name__=='__main__':main()
