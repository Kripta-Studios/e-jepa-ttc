"""Isolated CPU inference caches; no temporal output or optimizer is cached."""
from __future__ import annotations

import hashlib
from collections import OrderedDict
from pathlib import Path

import numpy as np
import torch


class FrameReadCache:
    """Bounded HDF5 RGB frame LRU keyed by immutable file identity and frame index."""

    def __init__(self, max_bytes=128 * 1024**2):
        self.max_bytes = max_bytes
        self.cache = OrderedDict()
        self.bytes = 0
        self.peak_bytes = 0
        self.hits = self.misses = 0

    def clear(self):
        self.cache.clear()
        self.bytes = 0

    def wrap(self, original):
        def getitem(dataset, index, *args, **kwargs):
            if dataset.name not in ('/blackflys/right/data', '/blackflys/left/data') or not isinstance(index, (int, np.integer)):
                return original(dataset, index, *args, **kwargs)
            if args or kwargs:
                return original(dataset, index, *args, **kwargs)
            path = Path(dataset.file.filename).resolve()
            stat = path.stat()
            key = (str(path), stat.st_size, stat.st_mtime_ns, dataset.name, int(index), str(dataset.dtype), tuple(dataset.shape))
            if key in self.cache:
                self.hits += 1
                value = self.cache.pop(key)
                self.cache[key] = value
                return value.copy()
            self.misses += 1
            value = original(dataset, index)
            if value.nbytes <= self.max_bytes:
                while self.cache and self.bytes + value.nbytes > self.max_bytes:
                    _, old = self.cache.popitem(last=False)
                    self.bytes -= old.nbytes
                self.cache[key] = value.copy()
                self.bytes += value.nbytes
                self.peak_bytes = max(self.peak_bytes, self.bytes)
            return value
        return getitem


class EndpointCache:
    """Memoize the frozen endpoint encoder before all temporal smoothing.

    The lifetime is one complete query. The caller binds sensor/ROI/registration,
    source code, checkpoint and CPU precision, and sets actual frame timestamps
    for each call. Cache hits also require identical numerical tensor bytes.
    """

    def __init__(self, encoder, binding, max_bytes=64 * 1024**2):
        self.encoder = encoder
        self.original = encoder.forward
        self.binding = binding
        self.max_bytes = max_bytes
        self.timestamps = None
        self.cache = OrderedDict()
        self.bytes = self.peak_bytes = 0
        self.hits = self.misses = 0

    def forward(self, values, *, return_dense_features=False):
        if self.encoder.training or torch.is_grad_enabled():
            raise ValueError('Endpoint caching requires eval and disabled autograd')
        if values.device.type != 'cpu' or values.dtype != torch.float32:
            raise ValueError('Only the registered CPU FP32 route is admitted')
        if self.timestamps is None or len(self.timestamps) != len(values):
            raise ValueError('Actual frame identities required for every encoder input')
        keys = [(self.binding, int(ts), return_dense_features, tuple(v.shape), hashlib.sha256(v.contiguous().numpy().tobytes()).hexdigest()) for ts, v in zip(self.timestamps, values, strict=True)]
        missing = list(dict.fromkeys(k for k in keys if k not in self.cache))
        indices = [keys.index(k) for k in missing]
        if missing:
            result = self.original(values[indices], return_dense_features=return_dense_features)
            for i, key in enumerate(missing):
                item = tuple(None if v is None else v[i:i+1].clone() for v in result)
                size = sum(v.numel() * v.element_size() for v in item if v is not None)
                if self.bytes + size > self.max_bytes:
                    raise MemoryError('Per-query endpoint cache budget exceeded')
                self.cache[key] = item
                self.bytes += size
                self.peak_bytes = max(self.peak_bytes, self.bytes)
        self.misses += len(missing)
        self.hits += len(keys) - len(missing)
        first = self.cache[keys[0]]
        return tuple(None if v is None else torch.cat([self.cache[k][i] for k in keys], dim=0) for i, v in enumerate(first))


class DenseMapCache(EndpointCache):
    """Reuse only CNN maps; preserve native batch shape in token/foreground heads."""

    def forward(self, values, *, return_dense_features=False):
        if self.encoder.training or torch.is_grad_enabled() or values.device.type != 'cpu' or values.dtype != torch.float32:
            raise ValueError('Dense caching requires CPU FP32 eval without autograd')
        if self.timestamps is None or len(self.timestamps) != len(values):
            raise ValueError('Frame timestamps required')
        keys = [(self.binding, int(ts), tuple(v.shape), hashlib.sha256(v.contiguous().numpy().tobytes()).hexdigest()) for ts, v in zip(self.timestamps, values, strict=True)]
        missing = list(dict.fromkeys(k for k in keys if k not in self.cache))
        if missing:
            dense = self.encoder.features(values[[keys.index(k) for k in missing]])
            for i, key in enumerate(missing):
                item = dense[i:i+1].clone()
                size = item.numel()*item.element_size()
                if self.bytes+size > self.max_bytes: raise MemoryError('Dense cache limit')
                self.cache[key] = item
                self.bytes += size
                self.peak_bytes = max(self.peak_bytes, self.bytes)
        self.misses += len(missing)
        self.hits += len(keys)-len(missing)
        features = torch.cat([self.cache[k] for k in keys])
        foreground_input = values if self.encoder.foreground_from_input else features
        return self.encoder.foreground(foreground_input), self.encoder.token(features), features if return_dense_features else None
