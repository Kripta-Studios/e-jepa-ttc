"""Regenerate per-frame distributions without dividing history percentiles by seven."""

from __future__ import annotations

import argparse
import hashlib
import json
import math
import statistics
from pathlib import Path


def distribution(values: list[float]) -> dict[str, float | int]:
    ordered = sorted(values)
    if not ordered or not all(math.isfinite(value) for value in ordered):
        raise ValueError("Nonempty finite frame measurements required")

    def quantile(fraction: float) -> float:
        position = (len(ordered) - 1) * fraction
        low = math.floor(position)
        high = min(low + 1, len(ordered) - 1)
        return ordered[low] + (ordered[high] - ordered[low]) * (position - low)

    return {
        "count": len(ordered),
        "mean_ms": statistics.fmean(ordered),
        "median_ms": quantile(0.5),
        "p95_ms": quantile(0.95),
        "p99_ms": quantile(0.99),
        "min_ms": ordered[0],
        "max_ms": ordered[-1],
    }


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument(
        "--source", type=Path,
        default=Path("artifacts/train40_runtime_council_20261028/frame_sealing_cpu/idx553_v1/summary.json"),
    )
    parser.add_argument(
        "--output", type=Path,
        default=Path("docs/train40_runtime_council_20261028/FRAME_SEALING_CPU_GENERATED.md"),
    )
    args = parser.parse_args()
    source_bytes = args.source.read_bytes()
    report = json.loads(source_bytes)
    assert report["status"] == "COMPLETE" and report["all_parity_passed"]
    assert report["preservation"]["passed"] and report["raw_stat_preserved"]
    measured = [row for row in report["records"] if row["phase"] == "measured"]
    frames = [frame for row in measured for frame in row["frame_checks"]]
    assert len(measured) == 20 and len(frames) == 140
    assert {tuple(frame["shape"]) for frame in frames} == {(1200, 1920, 3)}
    assert {frame["bytes"] for frame in frames} == {6_912_000}
    fields = {
        "copy_ms": "Copia C-order → bytes",
        "payload_sha256_ms": "SHA-256 del payload",
        "descriptor_ms": "Descriptor JSON + su SHA-256",
        "split_frame_ms": "Suma de los tres intervalos instrumentados",
        "original_factory_reported_ms": "Intervalo interno informado por freeze original",
    }
    result = {
        "source": args.source.as_posix(),
        "source_sha256": hashlib.sha256(source_bytes).hexdigest(),
        "scope": "MEASURED_FRAME_CHECKS_140_NOT_HISTORY_PERCENTILES_DIVIDED_BY_7",
        "warmup_pairs_preserved_but_not_in_this_population": sum(
            row["phase"] == "warmup" for row in report["records"]
        ),
        "measured_pairs": len(measured),
        "measured_frames": len(frames),
        "quantile_method": "linear",
        "per_frame": {
            field: distribution([frame[field] for frame in frames]) for field in fields
        },
        "by_frame_id": {
            str(frame_id): {
                field: distribution([
                    frame[field] for frame in frames if frame["frame_id"] == frame_id
                ])
                for field in fields
            }
            for frame_id in sorted({frame["frame_id"] for frame in frames})
        },
        "measured_history_distributions": report["distributions"]["measured"],
        "runtime_v2_implemented": False,
        "sustained_50hz_or_complete_20ms_demonstrated": False,
    }
    args.output.parent.mkdir(parents=True, exist_ok=True)
    args.output.with_suffix(".json").write_text(
        json.dumps(result, indent=2, allow_nan=False), encoding="utf-8"
    )
    table = [
        "| Tramo por frame, ms | n | Media | Mediana | p95 | p99 | Máximo |",
        "|---|---:|---:|---:|---:|---:|---:|",
    ]
    for field, label in fields.items():
        stats = result["per_frame"][field]
        table.append(
            f"| {label} | {stats['count']} | "
            + " | ".join(f"{stats[key]:.4f}" for key in (
                "mean_ms", "median_ms", "p95_ms", "p99_ms", "max_ms"
            )) + " |"
        )
    sha = result["per_frame"]["payload_sha256_ms"]
    copy = result["per_frame"]["copy_ms"]
    descriptor = result["per_frame"]["descriptor_ms"]
    history = report["distributions"]["measured"]
    text = f"""# Sellado RGB CPU: coste por frame medido

Generado por `artifacts/train40_runtime_council_20261028/frame_sealing_cpu/write_findings.py`.
Fuente: [recibo original](../../{args.source.as_posix()});
[distribuciones y SHA de origen](FRAME_SEALING_CPU_GENERATED.json).

Se conservaron {len(report['records'])} pares de siete frames, incluidos dos warmups.
Todas las rutas pasan bytes, píxeles, SHA del payload, descriptor, IDs y clocks;
{report['preservation']['files']} archivos y el stat RAW permanecen intactos.
La tabla utiliza directamente los **140 `frame_checks` de los 20 pares medidos**,
con órdenes AB/BA alternados. No divide percentiles del historial entre siete.
Cada frame contiene 6.912.000 bytes uint8, shape 1200×1920×3, strides [5760,3,1].

{chr(10).join(table)}

El intervalo interno del factory se mide antes de publicar el token en su
registro; no equivale a un reloj externo completo por frame. Los tres intervalos
manuales excluyen validación, registro y publicación del token. El JSON conserva
también las distribuciones por ID de frame y las del historial completo.

## Interpretación y margen potencial

La SHA del payload representa trabajo material: media {sha['mean_ms']:.4f} ms
por frame, frente a {copy['mean_ms']:.4f} ms de copia y {descriptor['mean_ms']:.4f} ms
de descriptor. En los siete frames completos, las medias fueron
{history['original_ms']['mean_ms']:.4f} ms para el factory original,
{history['payload_sha256_ms']['mean_ms']:.4f} ms de SHA y
{history['copy_ms']['mean_ms']:.4f} ms de copia.

La hipótesis inicial de eliminar el hash del payload mediante identidad local
se ha sustituido por una opción que conserva identidad fuerte de contenido:
[micro SHA-256 frente a BLAKE3-256](FRAME_HASH_CPU_GENERATED.md) y
[diseño V2 propuesto](FROZEN_RGB_V2_DESIGN.md). Los
{sha['mean_ms']:.4f} ms medios y p95 {sha['p95_ms']:.4f} ms aquí son costes
observados de SHA, **no una mejora V2 medida ni un plazo garantizado**.
Un update sin frames nuevos no paga este SHA;
los bursts y el número real de llegadas deben conservarse en el ensayo completo.
No se puede restar este percentil al p95 de otro ensayo o convertir su inversa
en frecuencia sostenida.

V2 no está implementada. Los tokens V1 y su integridad permanecen intactos.
Una futura V2 debe distinguir algoritmo/namespace y conservar copia inmutable,
procedencia autenticada y validación de los píxeles realmente consumidos, sin
rebajar la paridad TTC ni los snapshots de pesos/normalizadores. El coste de
copia, hash fuerte, metadata, preparación, transferencias y modelos seguirá
dentro del deadline.

## Límites

Es un microbenchmark CPU sobre el mismo historial ya disponible, con lecturas,
validación y escritura de recibos fuera del reloj. No simula llegada física,
colas ni solapamiento GPU. Las 140 muestras comparten 20 pares y siete frames:
no son 140 episodios independientes ni una base para un intervalo estadístico
de producción. No se ha implementado runtime nuevo ni demostrado 50 Hz/20 ms.
"""
    args.output.write_text(text, encoding="utf-8")


if __name__ == "__main__":
    main()
