"""Regenerate this experiment's narrative and timing table from saved evidence."""

import json
from pathlib import Path

root = Path(__file__).resolve().parent
report = json.loads((root / "gpu_sensitive_v2/summary.json").read_text(encoding="utf-8"))
lines = [
    "# Compiled complete transport: GPU findings",
    "",
    "The custom dispatcher integration works, but the default compiler variant is **not admitted**: "
    "5/7 query IDs pass all three fixed TTC gates. All seven final PHASE outputs pass. "
    "R_CTX fails for query 488 and E_CTX fails for query 505. No tolerance was changed.",
    "",
    "Timing covers prepared GPU producer + PAIR + PHASE17 + guards only. It excludes raw "
    "preprocessing, H2D and final context. It does not establish complete-update latency or "
    "sustained 50 Hz.",
    "",
    "| Query | Original graph mean ms | Compiled graph mean ms | Compiled p95 ms | All 3 TTC pass |",
    "|---:|---:|---:|---:|:---:|",
]
for row in report["records"]:
    original = row["latency_ms"]["original_graph"]
    compiled = row["latency_ms"]["compiled_transport_graph"]
    lines.append(
        f"| {row['index']} | {original['mean_ms']:.3f} | {compiled['mean_ms']:.3f} | "
        f"{compiled['p95_ms']:.3f} | {row['parity']['compiled_transport_graph']['passed']} |"
    )
lines += [
    "",
    "This table is generated by write_findings.py from gpu_sensitive_v2/summary.json, "
    "which includes every measured latency sample and TTC output.",
    "",
    f"Preservation: {report['preservation']['files']} files checked, "
    f"passed={report['preservation']['passed']}; loaded model weight values "
    f"unchanged={report['weights_unchanged']}.",
    "",
    "The eager oracle for each primary ROI matches its preserved prediction. Coverage and "
    "history metadata pass for all measured queries. Intermediate feature comparisons use "
    "the same tolerance as diagnostics; they do not replace or strengthen the final TTC gate.",
    "",
    "## Compiler execution evidence",
    "",
    "Each modality has three opaque dispatcher nodes in traced submodule graphs: A5 fine, "
    "C2F fine/coarse. Six compiled dispatcher invocations per active block occurred during "
    "initial evaluation, tracing, warmup and capture. No dispatcher callback occurred during "
    "timed CUDA Graph replays.",
    "",
    "The parent inlined_graph view contains zero operator nodes in this PyTorch build; "
    "recursive submodule IR contains the nodes. Both views are preserved in *_jit_ir.txt. "
    "Initial gpu_idx0_v1 measured correct query-0 parity, but its final audit wrongly required "
    "the node in the parent inlined view. That failed result is preserved. Version 2 audits "
    "both views and preserves the full IR.",
    "",
    "Executed source bytes are in gpu_sensitive_v2/sources/ with SHA bindings; later edits "
    "do not overwrite that evidence.",
    "",
    "## Next controlled variants",
    "",
    "--precision-mode no-fma adds emulate_precision_casts=True. "
    "--precision-mode no-fma-exact-div also adds eager_numerics.division_rounding=True. "
    "Both variants were subsequently measured and rejected; see PRECISION_VARIANTS.md.",
    "",
    "Use --compiler-cache artifacts/train40_runtime_resume_20261027/compiled_pipeline/"
    "gpu_sensitive_v2 to reuse valid compiler keys. The driver records cache byte inventories "
    "before/after and excludes setup from warm latency. A precision option changes compiler "
    "keys; it does not guarantee the original ATen reduction order.",
    "",
    "Do not promote a twenty-ROI run until all three TTC outputs pass the sensitive queries. "
    "The original graph runtime remains the admitted path.",
    "",
    "The opacity and mutation contract follows the official "
    "[PyTorch custom operator guide](https://docs.pytorch.org/tutorials/advanced/python_custom_ops). "
    "This wrapper supports JIT-to-CUDA-Graph inference. It does not claim standalone graph "
    "serialization, outer Dynamo/export, autograd or vmap support.",
]
(root / "GPU_FINDINGS.md").write_text("\n".join(lines) + "\n", encoding="utf-8")

variants = [
    ("default", "gpu_sensitive_v2"),
    ("no-fma", "gpu_no_fma_v3"),
    ("no-fma-exact-div", "gpu_no_fma_exact_div_v4"),
]
precision = [
    "# Compiler precision variants: measured rejection",
    "",
    "Every row below is generated from its saved summary.json. One real primary ROI was "
    "tested at seven sensitive queries; no two-object compiled run was promoted. "
    "The fixed gate is all three TTC outputs with atol=rtol=1e-5. "
    "Intermediate features are diagnostics only. Every final PHASE output passed.",
    "",
    "| Variant | All-three-TTC queries passed | Failed outputs | Files preserved |",
    "|---|---:|---|---:|",
]
for label, directory in variants:
    data = json.loads((root / directory / "summary.json").read_text(encoding="utf-8"))
    records = data["records"]
    passed = sum(row["parity"]["compiled_transport_graph"]["passed"] for row in records)
    failures = []
    for row in records:
        for component, comparison in row["parity"]["compiled_transport_graph"]["per_query"][0]["components"].items():
            if not comparison["passed"]:
                failures.append(
                    f"{row['index']} {component}: abs {comparison['abs_error']:.9g} "
                    f"> allowed {comparison['allowed_abs_error']:.9g}"
                )
    precision.append(
        f"| {label} | {passed}/{len(records)} | {'; '.join(failures)} | "
        f"{data['preservation']['files']} |"
    )
precision += [
    "",
    "Every variant preserved loaded weight values and passed its file snapshot. "
    "Their prepared GPU producer latencies were approximately 10 ms, but raw preparation, "
    "H2D and final context were outside that timing. These rejected variants do not establish "
    "complete-update 50 Hz.",
    "",
    "The next native-ATen entropy-log boundary is implemented behind --native-log. "
    "It has targeted CPU validation only; no Inductor/CUDA/final-TTC admission has been run. "
    "The next driver also records feature errors separately for E_A5_MATCHED, E_C2F_MATCHED, "
    "R_A5 and R_C2F, under matched encoder reuse and batch shapes.",
]
(root / "PRECISION_VARIANTS.md").write_text("\n".join(precision) + "\n", encoding="utf-8")
