"""Compare identical RAW updates with eager and compiled integer prefix selectors.

The output boundary is all requested event tensors on the device. Models, TTC,
disk replay and synthetic startup compilation/capture are excluded and reported
separately. Additional ROIs are capacity regions, not annotated object tracks.
"""

from __future__ import annotations

import argparse
import gc
import hashlib
import json
import platform
import subprocess
import sys
import time
from pathlib import Path
from typing import Any

ROOT = Path(__file__).resolve().parents[1]
sys.path[:0] = [str(ROOT / "src"), str(ROOT)]


def run(args: argparse.Namespace) -> None:
    """Preserve paired timings, parity and matching source bytes on failure too."""
    if (
        args.steps < 1
        or args.warmup < 0
        or any(n < 1 or n > 20 for n in args.objects)
        or len(set(args.objects)) != len(args.objects)
    ):
        raise ValueError("positive steps, nonnegative warmup, unique object counts1..20 required")
    if not args.output.resolve().is_relative_to((ROOT / "artifacts").resolve()):
        raise ValueError("output must be a fresh directory under this repo's artifacts")
    reference_bytes = (
        (args.steps + args.warmup) * sum(args.objects) * 4 * 3 * 12 * args.roi_size**2 * 4
    )
    if reference_bytes > args.max_reference_bytes:
        raise ValueError("reference tensor storage exceeds the explicit host memory budget")
    args.output.mkdir(parents=True, exist_ok=False)
    if args.compile_backend == "inductor":
        if args.compiler_deps is None:
            raise ValueError("Inductor requires an explicitly isolated --compiler-deps directory")
        from scripts.benchmark_train40_transport import configure_isolated_compiler

        configure_isolated_compiler(args.compiler_deps, args.output / "compiler_cache")
    import numpy as np
    import torch

    from operational.evttc_transfer.inputs import EvTTCEventReader
    from operational.train40_runtime.compiled_prefix import CompiledTemporalPrefixEventHistory
    from operational.train40_runtime.measurement import latency_distribution
    from operational.train40_runtime.temporal_prefix import TemporalPrefixEventHistory
    from scripts.benchmark_train40_prefix import _canonical, _rois, _sha, _slice, _sync, _write

    torch.set_num_threads(2)
    torch.backends.cuda.matmul.allow_tf32 = False
    torch.backends.cudnn.allow_tf32 = False
    torch.backends.cudnn.benchmark = False
    sources = [
        Path(__file__),
        ROOT / "scripts/benchmark_train40_prefix.py",
        ROOT / "scripts/benchmark_train40_transport.py",
        ROOT / "operational/train40_runtime/compiled_prefix.py",
        ROOT / "operational/train40_runtime/temporal_prefix.py",
        ROOT / "operational/train40_runtime/integral.py",
        ROOT / "operational/train40_runtime/normalization.py",
    ]
    source_hashes = {str(path.relative_to(ROOT)): _sha(path) for path in sources}
    for source in sources:
        target = args.output / "sources" / Path(str(source.relative_to(ROOT)) + ".txt")
        target.parent.mkdir(parents=True, exist_ok=True)
        target.write_bytes(source.read_bytes())
    updates = args.steps + args.warmup
    if args.synthetic:
        start = 738
        anchor = start + 600_000 + updates * 20_000
        rng = np.random.default_rng(7)
        events = (30 + updates) * args.packet_events
        raw = {
            "x": rng.integers(0, args.width, events, dtype=np.int32),
            "y": rng.integers(0, args.height, events, dtype=np.int32),
            "t": np.sort(rng.integers(start, anchor, events, dtype=np.int64)),
            "p": rng.choice(np.array([-1, 1], dtype=np.int8), events),
        }
        primary = [0.0, 0.0, float(args.width), float(args.height)]
        data_binding: dict[str, Any] = {"synthetic": True, "seed": 7}
    else:
        if args.manifest is None:
            raise ValueError("--manifest required without --synthetic")
        row = json.loads(args.manifest.read_text(encoding="utf-8"))["rows"][args.index]
        anchor = int(row["anchor_us"])
        start = anchor - 600_000 - updates * 20_000
        if start < 0:
            raise ValueError("requested updates extend before a nonnegative sensor clock")
        reader = EvTTCEventReader(Path(row["raw_path"])).open()
        try:
            raw = reader.read_window(start, anchor)
        finally:
            reader.close()
        primary = row["square_xyxy"]
        data_binding = {
            "synthetic": False,
            "manifest_sha256": _sha(args.manifest),
            "query_id": row["query_id"],
            "index": args.index,
            "raw_path": row["raw_path"],
            "extra_rois": "SYNTHETIC_CAPACITY_REGIONS_NOT_ANNOTATED_TRACKS",
        }
    initial_end = start + 600_000
    initial = _slice(raw, start, initial_end)
    packets = [
        _slice(raw, initial_end + i * 20_000, initial_end + (i + 1) * 20_000)
        for i in range(updates)
    ]
    binding = {
        "scope": "RAW_CPU_PACKET_TO_ALL_EVENT_TENSORS_NO_MODELS_OR_TTC",
        "data": data_binding,
        "sources": source_hashes,
        "source_commit": subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip(),
        "python": platform.python_version(),
        "torch": torch.__version__,
        "cuda": torch.version.cuda,
        "device": args.device,
        "gpu": torch.cuda.get_device_name() if torch.device(args.device).type == "cuda" else None,
        "normalization": args.normalization,
        "graphs": args.graphs,
        "compile_backend": args.compile_backend,
        "source_timestamps_us": [start, initial_end, anchor],
        "objects": args.objects,
        "measured_updates": args.steps,
        "warmup_updates": args.warmup,
        "reference_host_bytes": reference_bytes,
        "packet_events": [len(packet["t"]) for packet in packets],
        "input_sha256": {
            key: hashlib.sha256(value.tobytes()).hexdigest() for key, value in raw.items()
        },
        "target_50hz_demonstrated": False,
    }
    _write(args.output / "binding.json", binding)
    records: list[dict[str, Any]] = []
    reference: dict[int, list[torch.Tensor]] = {}
    errors: list[str] = []
    caught: BaseException | None = None
    try:
        for cls in (TemporalPrefixEventHistory, CompiledTemporalPrefixEventHistory):
            for objects in args.objects:
                is_compiled = cls is CompiledTemporalPrefixEventHistory
                options: dict[str, Any] = {}
                if is_compiled:
                    options["compile_backend"] = (
                        None if args.compile_backend == "none" else args.compile_backend
                    )
                if torch.device(args.device).type == "cuda":
                    torch.cuda.reset_peak_memory_stats(args.device)
                history = cls(
                    device=args.device,
                    width=args.width,
                    height=args.height,
                    roi_size=args.roi_size,
                    max_packet_events=args.max_packet_events,
                    max_objects=objects,
                    roi_batch_size=min(objects, args.roi_batch_size),
                    normalization=args.normalization,
                    graphs=args.graphs,
                    **options,
                )
                prime_start = time.perf_counter()
                if args.graphs:
                    history.prime_query_graphs([objects])
                elif isinstance(history, CompiledTemporalPrefixEventHistory):
                    history.prime_selectors([objects])
                _sync(args.device)
                priming_ms = (time.perf_counter() - prime_start) * 1000
                initial_start = time.perf_counter()
                history.initialize(initial, start, initial_end)
                _sync(args.device)
                initialization_ms = (time.perf_counter() - initial_start) * 1000
                rois = _rois(primary, objects)
                measured = []
                outputs: list[torch.Tensor] = []
                exact, max_abs = True, 0.0
                for step, packet in enumerate(packets):
                    left = initial_end + step * 20_000
                    _sync(args.device)
                    begin_ns = time.perf_counter_ns()
                    history.ingest(packet, left, left + 20_000)
                    output = history.query_many(rois)
                    _sync(args.device)
                    elapsed_ms = (time.perf_counter_ns() - begin_ns) / 1e6
                    actual = output.cpu().clone()
                    if is_compiled:
                        expected = reference[objects][step]
                        exact &= torch.equal(actual, expected)
                        max_abs = max(max_abs, float((actual - expected).abs().max()))
                    else:
                        outputs.append(actual)
                    if step >= args.warmup:
                        measured.append(elapsed_ms)
                    print(
                        json.dumps(
                            {
                                "backend": cls.__name__,
                                "objects": objects,
                                "step": step,
                                "ms": elapsed_ms,
                            }
                        ),
                        flush=True,
                    )
                if not is_compiled:
                    reference[objects] = outputs
                canonical = _canonical(raw, anchor, tuple(rois[0]), args.roi_size)
                canonical_max_abs = float((actual[0] - canonical).abs().max())
                canonical_passed = bool(torch.allclose(actual[0], canonical, rtol=1e-5, atol=1e-5))
                if not exact or not canonical_passed:
                    errors.append(f"{cls.__name__}/{objects}: parity failed")
                records.append(
                    {
                        "backend": cls.__name__,
                        "objects": objects,
                        "samples_ms": measured,
                        "latency_ms": latency_distribution(measured),
                        "synthetic_priming_ms_excluded": priming_ms,
                        "initialization_ms_excluded": initialization_ms,
                        "compiled_vs_prefix_bit_exact": exact if is_compiled else None,
                        "compiled_vs_prefix_max_abs": max_abs if is_compiled else None,
                        "canonical_primary_final_max_abs": canonical_max_abs,
                        "canonical_primary_final_passed": canonical_passed,
                        "stats": history.stats,
                        "process_peak_allocated_bytes": torch.cuda.max_memory_allocated(args.device)
                        if torch.device(args.device).type == "cuda"
                        else None,
                        "process_peak_reserved_bytes": torch.cuda.max_memory_reserved(args.device)
                        if torch.device(args.device).type == "cuda"
                        else None,
                    }
                )
                _write(args.output / "results.json", records)
                del history, output, actual
                gc.collect()
                if torch.device(args.device).type == "cuda":
                    torch.cuda.empty_cache()
    except BaseException as exc:
        caught = exc
        errors.append(f"{type(exc).__name__}: {exc}")
    finally:
        unchanged = all(
            _sha(path) == source_hashes[str(path.relative_to(ROOT))] for path in sources
        )
        if not unchanged:
            errors.append("source changed during benchmark")
        _write(
            args.output / "summary.json",
            {
                "status": "FAILED" if errors else "COMPLETE",
                "errors": errors,
                "sources_unchanged": unchanged,
                "results": records,
                "target_50hz_demonstrated": False,
            },
        )
    if caught:
        raise caught
    if errors:
        raise RuntimeError("compiled prefix benchmark failed; inspect summary.json")


def main() -> None:
    """Parse an explicit isolated-compiler representation experiment."""
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--output", type=Path, required=True)
    parser.add_argument("--manifest", type=Path)
    parser.add_argument("--index", type=int, default=553)
    parser.add_argument("--objects", type=int, nargs="+", default=[1, 20])
    parser.add_argument("--steps", type=int, default=5)
    parser.add_argument("--warmup", type=int, default=2)
    parser.add_argument("--device", default="cuda")
    parser.add_argument("--width", type=int, default=1280)
    parser.add_argument("--height", type=int, default=720)
    parser.add_argument("--roi-size", type=int, default=128)
    parser.add_argument("--roi-batch-size", type=int, default=4)
    parser.add_argument("--max-packet-events", type=int, default=2_000_000)
    parser.add_argument("--max-reference-bytes", type=int, default=2 * 1024**3)
    parser.add_argument("--normalization", choices=("sort", "topk"), default="topk")
    parser.add_argument("--graphs", action=argparse.BooleanOptionalAction, default=True)
    parser.add_argument(
        "--compile-backend", choices=("inductor", "eager", "none"), default="inductor"
    )
    parser.add_argument("--compiler-deps", type=Path)
    parser.add_argument("--synthetic", action="store_true")
    parser.add_argument("--packet-events", type=int, default=200)
    run(parser.parse_args())


if __name__ == "__main__":
    main()
